
### 1. KNN from Scratch vs. Scikit-Learn

Implement KNN from scratch using Euclidean distance and majority vote, and verify with `KNeighborsClassifier`.

```python
import numpy as np
from collections import Counter
from sklearn.neighbors import KNeighborsClassifier

X = np.array([[1, 2], [2, 3], [3, 1], [6, 5], [7, 7], [8, 6]])
y = np.array([0, 0, 0, 1, 1, 1])
X_new = np.array([[2, 2], [7, 5]])

# From Scratch (k=3)
preds_scratch = []
for p in X_new:
    dists = np.sqrt(np.sum((X - p) ** 2, axis=1))
    nearest_labels = y[np.argsort(dists)[:3]]
    preds_scratch.append(Counter(nearest_labels).most_common(1)[0][0])

# Scikit-Learn
knn = KNeighborsClassifier(n_neighbors=3).fit(X, y)

print("Scratch:", preds_scratch)
print("Sklearn:", knn.predict(X_new).tolist())

```

### 2. Descriptive Statistics from Scratch vs. `statistics` Module

Compute Mean, Median, Mode, Variance, and Standard Deviation using custom formulas and verify with `statistics`.

```python
import statistics

d = [12, 15, 12, 18, 21, 24, 19, 15, 12, 27]
n = len(d)
s = sorted(d)

mean = sum(d) / n
median = (s[n//2 - 1] + s[n//2]) / 2 if n % 2 == 0 else s[n//2]
mode = max(set(d), key=d.count)
var = sum((x - mean) ** 2 for x in d) / (n - 1)
std = var ** 0.5

print("Custom :", round(mean, 2), median, mode, round(var, 2), round(std, 2))
print("Module :", round(statistics.mean(d), 2), statistics.median(d), statistics.mode(d), round(statistics.variance(d), 2), round(statistics.stdev(d), 2))

```

### 3. Multiple Linear Regression for House Price Prediction

Predict house prices from area, bedrooms, and age, report MAE, MSE, RMSE, and $R^2$, and predict for a new house.

```python
import numpy as np
from sklearn.linear_model import LinearRegression
```

### 4. K-Means Clustering & The Elbow Method

Apply K-Means on 2D data, plot clusters with centroids, and plot the Elbow Method.

```python
import matplotlib.pyplot as plt
from sklearn.datasets import make_blobs
from sklearn.cluster import KMeans

X, _ = make_blobs(n_samples=200, centers=4, random_state=42)

# 1. Elbow Plot
wcss = [KMeans(n_clusters=k, n_init=10, random_state=42).fit(X).inertia_ for k in range(1, 8)]
plt.plot(range(1, 8), wcss, marker='o')
plt.title("Elbow Method")
plt.show()

# 2. Cluster Plot (k=4)
km = KMeans(n_clusters=4, n_init=10, random_state=42).fit(X)
plt.scatter(X[:, 0], X[:, 1], c=km.labels_)
plt.scatter(km.cluster_centers_[:, 0], km.cluster_centers_[:, 1], c='red', marker='X', s=150)
plt.title("K-Means Clusters")
plt.show()

```

### 8. Model Comparison: Logistic Regression, KNN, and Decision Tree

Compare the three models using accuracy, precision, recall, and F1-score, and plot the accuracies.

```python
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import KNeighborsClassifier
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

X, y = load_iris(return_X_y=True)
models = {
    'LogReg': LogisticRegression(max_iter=200),
    'KNN': KNeighborsClassifier(),
    'Tree': DecisionTreeClassifier()
}

accs = []
for name, m in models.items():
    p = m.fit(X, y).predict(X)
    accs.append(accuracy_score(y, p))
    print(f"{name}: Acc={accuracy_score(y,p):.2f}, Prec={precision_score(y,p,average='macro'):.2f}, Rec={recall_score(y,p,average='macro'):.2f}, F1={f1_score(y,p,average='macro'):.2f}")

plt.bar(models.keys(), accs)
plt.title("Accuracy Comparison")
plt.show()

```

### 9. Simple Linear Regression from Scratch (Least-Squares)

Implement Simple Linear Regression using least-squares formulas, predict a new value, compute MSE and $R^2$, and plot.

```python
import numpy as np
import matplotlib.pyplot as plt

x = np.array([1, 2, 3, 4, 5], dtype=float)
y = np.array([3, 5, 7, 9, 11], dtype=float)

b1 = np.sum((x - x.mean()) * (y - y.mean())) / np.sum((x - x.mean()) ** 2)
b0 = y.mean() - b1 * x.mean()
y_pred = b0 + b1 * x

mse = np.mean((y - y_pred) ** 2)
r2 = 1 - np.sum((y - y_pred) ** 2) / np.sum((y - y.mean()) ** 2)

print(f"Slope={b1}, Intercept={b0}, MSE={mse:.2f}, R2={r2:.2f}")
print("Predict for x=6:", b0 + b1 * 6)

plt.scatter(x, y)
plt.plot(x, y_pred, color='red')
plt.show()

```

### 11. Multi-Class KNN with Feature Scaling & K-Tuning

Apply KNN on a 3-class dataset after scaling, report accuracy and confusion matrix, and plot accuracy vs. $k$.

```python
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score, confusion_matrix

X, y = load_iris(return_X_y=True)
X = StandardScaler().fit_transform(X)

ks = range(1, 10)
accs = [accuracy_score(y, KNeighborsClassifier(k).fit(X, y).predict(X)) for k in ks]

best_k = ks[accs.index(max(accs))]
pred = KNeighborsClassifier(best_k).fit(X, y).predict(X)

print("Best K:", best_k, "| Accuracy:", accuracy_score(y, pred))
print("Confusion Matrix:\n", confusion_matrix(y, pred))

plt.plot(ks, accs, marker='o')
plt.xlabel('K')
plt.ylabel('Accuracy')
plt.show()

```

### 16. Decision Tree Hyperparameter Tuning (`max_depth` & Criteria)

Tune `max_depth` by plotting training vs. testing accuracy, and compare `gini` and `entropy`.

```python
import matplotlib.pyplot as plt
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier

X, y = load_breast_cancer(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(X, y, random_state=42)

depths = range(1, 8)
tr_acc = [DecisionTreeClassifier(max_depth=d).fit(X_tr, y_tr).score(X_tr, y_tr) for d in depths]
te_acc = [DecisionTreeClassifier(max_depth=d).fit(X_tr, y_tr).score(X_te, y_te) for d in depths]

print("Gini Acc   :", DecisionTreeClassifier(criterion='gini').fit(X_tr, y_tr).score(X_te, y_te))
print("Entropy Acc:", DecisionTreeClassifier(criterion='entropy').fit(X_tr, y_tr).score(X_te, y_te))

plt.plot(depths, tr_acc, label='Train')
plt.plot(depths, te_acc, label='Test')
plt.legend()
plt.show()

```
